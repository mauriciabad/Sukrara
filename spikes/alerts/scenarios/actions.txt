# The alert notification's actions (SPEC.md §2): Mute <n> min, Mute for… (typed minutes), Slow down. They act on the
# episode of the notification's alert and end with it. Format: SPEC.md §7.

scenario Mute quiets the low for its minutes; at once when it ends
 0  glucose 65   => Low glucose
 5               => Low glucose
 7  mute low     => clear low                 # the notification's Mute 30 min
10                                            # would have been the repeat
37               => Low glucose               # still low when the mute ends: at once
42               => Low glucose
43

scenario Mute for… with typed minutes
settings mute.min=20
 0  glucose 65   => Low glucose
 2  mute low 10  => clear low                 # the typed minutes, not the setting's 20
12               => Low glucose
17               => Low glucose
18

scenario Slow down: the repeats come at the slower interval
 0  glucose 65   => Low glucose
 2  slow low                                  # the notification stays
 5                                            # would have been the repeat
15               => Low glucose               # 15 min (the setting) after the last alert
30               => Low glucose
31

scenario The mute ends with the episode: a new low alerts at once
 0  glucose 65   => Low glucose
 2  mute low     => clear low
 4  glucose 90                                # in range: the episode ends 5 min after the last low reading
12  glucose 65   => Low glucose               # a new episode
17               => Low glucose
18

scenario Slow down ends with the episode
 0  glucose 65   => Low glucose
 1  slow low
 3  glucose 90   => clear low
12  glucose 65   => Low glucose
17               => Low glucose               # the kind's own 5 min again
18

scenario A muted low still alerts when it turns very low; very low can't be muted
 0  glucose 65     => Low glucose
 2  mute low       => clear low
 5  glucose 50     => Very low glucose        # more severe: at once, the mute doesn't cover very low
10                 => Very low glucose
11  mute very_low                             # no effect
12  glucose 65     => clear very_low          # low again: still muted
15                                            # the very low's repeat would have been here
32                 => Low glucose             # the mute from minute 2 ends
33

scenario Very low can be slowed down
 0  glucose 50     => Very low glucose
 1  slow very_low
15                 => Very low glucose
30                 => Very low glucose
31

scenario Mute and carbs together: quiet until both have ended
 0  glucose 65   => Low glucose
 2  carbs 10     => clear low
 3  mute low 5                                # nothing showing, nothing to clear
17               => Low glucose · eat more    # the carbs' window (the longer one) ends
22               => Low glucose · eat more
23

scenario A mute survives a restart
 0  glucose 65   => Low glucose
 2  mute low 10  => clear low
 5  restart                                   # nothing was showing: nothing to resume
12               => Low glucose
13

scenario Mute on a high; extremely high still alerts
 0  glucose 200  => High glucose
